   417→function TradeDialog({ listing, onClose, onSuccess }: {
   418→  listing: Listing
   419→  onClose: () => void
   420→  onSuccess: () => void
   421→}) {
   422→  const { user, setView } = useAppStore()
   423→  const { toast } = useToast()
   424→  const [amount, setAmount] = useState('')
   425→  const [paymentMethod, setPaymentMethod] = useState(listing.paymentMethods[0])
   426→  const [loading, setLoading] = useState(false)
   427→  const [compressing, setCompressing] = useState(false)
   428→  const [paymentScreenshot, setPaymentScreenshot] = useState<{ data: string; name: string; size: number } | null>(null)
   429→  const fileScreenshotRef = useRef<HTMLInputElement>(null)
   430→
   431→  const amountNum = parseFloat(amount) || 0
   432→  const fiatTotal = amountNum * listing.price
   433→  const minCrypto = listing.minOrder / listing.price
   434→  const maxCrypto = Math.min(listing.available, listing.maxOrder / listing.price)
   435→  const isBuying = listing.side === 'SELL' // SELL listing means buyer is buying
   436→
   437→  // Handle screenshot upload with compression
   438→  const handleScreenshotUpload = async (file: File) => {
   439→    if (!file) return
   440→    if (!file.type.startsWith('image/')) {
   441→      toast({ title: 'Invalid file', description: 'Please upload an image file', variant: 'destructive' })
   442→      return
   443→    }
   444→    if (file.size > 20 * 1024 * 1024) {
   445→      toast({ title: 'File too large', description: 'Maximum 20MB', variant: 'destructive' })
   446→      return
   447→    }
   448→    setCompressing(true)
   449→    try {
   450→      const compressed = await compressImageToBase64(file, {
   451→        maxWidth: 1600, maxHeight: 1600, quality: 0.8, mimeType: 'image/jpeg',
   452→      })
   453→      const size = Math.round((compressed.length - 'data:image/jpeg;base64,'.length) * 0.75)
   454→      setPaymentScreenshot({ data: compressed, name: file.name, size })
   455→      toast({ title: 'Screenshot uploaded', description: `${formatFileSize(size)} · ready to submit` })
   456→    } catch (e: any) {
   457→      toast({ title: 'Upload failed', description: e.message, variant: 'destructive' })
   458→    } finally {
   459→      setCompressing(false)
   460→    }
   461→  }
   462→
   463→  const submit = async () => {
   464→    if (!user) {
   465→      toast({ title: 'Please log in first', variant: 'destructive' })
   466→      setView('home')
   467→      return
   468→    }
   469→    if (amountNum <= 0) {
   470→      toast({ title: 'Enter valid amount', variant: 'destructive' })
   471→      return
   472→    }
   473→    if (amountNum < minCrypto) {
   474→      toast({ title: `Minimum is ${formatQty(minCrypto)} ${listing.asset}`, variant: 'destructive' })
   475→      return
   476→    }
   477→    if (amountNum > maxCrypto) {
   478→      toast({ title: `Maximum is ${formatQty(maxCrypto)} ${listing.asset}`, variant: 'destructive' })
   479→      return
   480→    }
   481→    // Require payment screenshot for buy orders
   482→    if (isBuying && !paymentScreenshot) {
   483→      toast({ title: 'Payment screenshot required', description: 'Upload proof of payment before placing the order', variant: 'destructive' })
   484→      return
   485→    }
   486→    setLoading(true)
   487→    try {
   488→      const res = await fetch('/api/p2p/orders', {
   489→        method: 'POST',
   490→        headers: { 'Content-Type': 'application/json' },
   491→        body: JSON.stringify({
   492→          listingId: listing.id,
   493→          amount: amountNum,
   494→          paymentMethod,
   495→          paymentScreenshot: paymentScreenshot?.data,
   496→        }),

  1001→  useEffect(() => {

  1003→      const filtered = prev.filter(m => availableMethods.includes(m))
  1004→      return filtered.length > 0 ? filtered : [availableMethods[0]]

  1006→  }, [fiat])
