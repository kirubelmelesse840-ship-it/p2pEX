   256→   256→function AuthButtons() {
   257→   257→  const [showAuth, setShowAuth] = useState<'login' | 'signup' | null>(null)
   258→   258→  const [showGoogle, setShowGoogle] = useState(false)
   259→   259→  const { toast } = useToast()
   260→   260→  const setUser = useAppStore(s => s.setUser)
   261→   261→  const [email, setEmail] = useState('')
   262→   262→  const [password, setPassword] = useState('')
   263→   263→  const [confirmPassword, setConfirmPassword] = useState('')
   264→   264→  const [loading, setLoading] = useState(false)
   265→   265→
   266→   266→  const submit = async (mode: 'login' | 'signup') => {
   267→   267→    if (mode === 'signup') {
   268→   268→      if (password.length < 6) {
   269→   269→        toast({ title: 'Password too short', description: 'At least 6 characters required', variant: 'destructive' })
   270→   270→        return
   271→   271→      }
   272→   272→      if (password !== confirmPassword) {
   273→   273→        toast({ title: 'Passwords do not match', variant: 'destructive' })
   274→   274→        return
   275→   275→      }
   276→   276→    }
   277→   277→    setLoading(true)
   278→   278→    try {
   279→   279→      const url = mode === 'login' ? '/api/auth/login' : '/api/auth'
   280→   280→      const body = mode === 'login'
   281→   281→        ? { email, password }
   282→   282→        : { email, password, name: email.split('@')[0] }
   283→   283→      const res = await fetch(url, {
   284→   284→        method: 'POST',
   285→   285→        headers: { 'Content-Type': 'application/json' },
   286→   286→        body: JSON.stringify(body),
   287→   287→      })
   288→   288→      const data = await res.json()
   289→   289→      if (data.error) throw new Error(data.error)
   290→   290→      setUser(data.user)
   291→   291→      toast({
   292→   292→        title: mode === 'login' ? 'Welcome back!' : 'Account created!',
   293→   293→        description: `Logged in as ${data.user.name}`,
   294→   294→      })
   295→   295→      setShowAuth(null)
   296→   296→      setEmail(''); setPassword(''); setConfirmPassword('')
   297→   297→    } catch (e: any) {
   298→   298→      toast({ title: 'Authentication failed', description: e.message, variant: 'destructive' })

   300→   300→      setLoading(false)

   712→   712→  const [unreadCount, setUnreadCount] = useState(0)
   713→   713→  const [open, setOpen] = useState(false)
   714→   714→

   717→   717→      const res = await fetch('/api/notifications')
   718→   718→      const d = await res.json()
   719→   719→      if (d.error) return
   720→   720→      setNotifications(d.notifications || [])
   721→   721→      setUnreadCount(d.unreadCount || 0)

   723→   723→  }, [])

   726→   726→    load()
   727→   727→    const t = setInterval(load, 15000)
   728→   728→    return () => clearInterval(t)
   729→   729→  }, [load])
   730→   730→
   731→   731→  const markRead = async (id: string) => {
   732→   732→    try {
   733→   733→      await fetch('/api/notifications', {
   734→   734→        method: 'POST',
   735→   735→        headers: { 'Content-Type': 'application/json' },
   736→   736→        body: JSON.stringify({ notificationId: id }),

   739→   739→    } catch {}
   740→   740→  }

   744→   744→      await markRead(n.id)

   752→   752→    if (diff < 86400000) return Math.floor(diff / 3600000) + 'h ago'

   759→   759→      case 'warning': return 'text-yellow-500'
