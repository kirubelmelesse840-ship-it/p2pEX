   250→   250→function DashboardTab() {
   251→   251→  const [data, setData] = useState<any>(null)
   252→   252→  const [loading, setLoading] = useState(false)
   253→   253→  const [refreshing, setRefreshing] = useState(false)
   254→   254→  const [error, setError] = useState<string | null>(null)
   255→   255→
   256→   256→  const load = useCallback(async () => {
   257→   257→    // Manual refresh button should feel instant — show cached data immediately, then update
   258→   258→    const cached = sessionStorage.getItem('admin-stats')
   259→   259→    if (cached && !data) {
   260→   260→      try { setData(JSON.parse(cached)) } catch {}
   261→   261→    }
   262→   262→    try {
   263→   263→      const res = await fetch('/api/admin/stats')
   264→   264→      // Handle empty response body (happens when function cold-starts or times out)
   265→   265→      const text = await res.text()
   266→   266→      if (!text) {
   267→   267→        // Empty response — keep existing data, don't overwrite with an error
   268→   268→        if (!data) setError('Server returned empty response. Tap Refresh to try again.')
   269→   269→        return
   270→   270→      }
   271→   271→      const d = JSON.parse(text)
   272→   272→      if (d.error) throw new Error(d.error)
   273→   273→      setData(d)
   274→   274→      setError(null)
   275→   275→      sessionStorage.setItem('admin-stats', JSON.stringify(d))
   276→   276→    } catch (e: any) {
   277→   277→      console.error('[admin dashboard]', e)
   278→   278→      // Only set error if we don't have any data yet — otherwise keep showing the cached data
   279→   279→      if (!data) setError(e.message || 'Failed to load dashboard')
   280→   280→    } finally {
   281→   281→      setLoading(false)
   282→   282→      setRefreshing(false)
   283→   283→    }
   284→   284→  }, [data])
   285→   285→
   286→   286→  useEffect(() => {
   287→   287→    load() // Load once on mount — no auto-refresh (admin taps Refresh button manually)
   288→   288→  }, [load])
   289→   289→
   290→   290→  if (!data) {
   291→   291→    return <div className="text-center py-12 text-muted-foreground">Loading dashboard...</div>
   292→   292→  }
   293→   293→
   294→   294→  if (error && !data) {
   295→   295→    return (
   296→   296→      <div className="text-center py-12">
   297→   297→        <AlertTriangle className="h-12 w-12 mx-auto text-red-500 mb-3" />
   298→   298→        <p className="text-sm text-muted-foreground mb-2">Failed to load dashboard</p>
   299→   299→        <p className="text-xs text-muted-foreground mb-4">{error}</p>
   300→   300→        <Button onClick={load} size="sm">
   301→   301→          <RefreshCw className="h-3.5 w-3.5 mr-1.5" /> Retry
   302→   302→        </Button>
   303→   303→      </div>
   304→   304→    )
   305→   305→  }
   306→   306→
   307→   307→  if (!data) return null
   308→   308→
   309→   309→  const s = data.stats
   310→   310→
   311→   311→  return (
   312→   312→    <div className="space-y-4">
   313→   313→      {/* Top metric cards */}
   314→   314→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   315→   315→        <MetricCard
   316→   316→          icon={<Users className="h-5 w-5" />}
   317→   317→          label="Total Users"
   318→   318→          value={s.users.total}
   319→   319→          sub={`${s.users.active} active · ${s.users.banned} banned`}
   320→   320→          color="text-blue-500"
   321→   321→        />
   322→   322→        <MetricCard
   323→   323→          icon={<Activity className="h-5 w-5" />}
   324→   324→          label="24h Volume"
   325→   325→          value={formatUsd(s.trades.volume24hUsd)}
   326→   326→          sub={`${s.trades.last24h} trades in 24h`}
   327→   327→          color="text-green-500"
   328→   328→        />
   329→   329→        <MetricCard
   330→   330→          icon={<Wallet className="h-5 w-5" />}
   331→   331→          label="Total Balance"
   332→   332→          value={formatUsd(s.wallets.totalUsdValue)}
   333→   333→          sub={`${formatUsd(s.wallets.totalLockedValue)} locked in orders`}
   334→   334→          color="text-yellow-500"
   335→   335→        />
   336→   336→        <MetricCard
   337→   337→          icon={<Shield className="h-5 w-5" />}
   338→   338→          label="KYC Pending"
   339→   339→          value={s.users.kycPending}
   340→   340→          sub={`${s.users.admins} managers`}
   341→   341→          color="text-orange-500"
   342→   342→        />
   343→   343→      </div>

   347→   347→        <SmallStat label="Active Pairs" value={`${s.pairs.active}/${s.pairs.total}`} />
   348→   348→        <SmallStat label="Open Orders" value={s.orders.open} />

   524→   524→
   525→   525→  const load = useCallback(async () => {
   526→   526→    // Only show loading spinner on the FIRST load (not on every refresh tap)
   527→   527→    if (users.length === 0) setLoading(true)
   528→   528→    try {

   530→   530→      if (search) params.set('search', search)

   532→   532→      if (kycFilter !== 'all') params.set('kyc', kycFilter)

   536→   536→      const res = await fetch(`/api/admin/users?${params}`, { signal: controller.signal })

   541→   541→        // Empty response — keep existing data, don't show error on subsequent refreshes

   543→   543→        return
