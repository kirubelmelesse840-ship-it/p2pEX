   428→// =================== USERS TAB ===================
   429→function UsersTab() {
   430→  const { toast } = useToast()
   431→  const [users, setUsers] = useState<any[]>([])
   432→  const [total, setTotal] = useState(0)
   433→  const [loading, setLoading] = useState(false)
   434→  const [refreshing, setRefreshing] = useState(false)
   435→  const [search, setSearch] = useState('')
   436→  const [statusFilter, setStatusFilter] = useState('all')
   437→  const [kycFilter, setKycFilter] = useState('all')
   438→  const [walletDialog, setWalletDialog] = useState<any>(null)
   439→  const [banDialog, setBanDialog] = useState<any>(null)
   440→  const [docDialog, setDocDialog] = useState<any>(null)
   441→
   442→  const load = useCallback(async () => {
   443→    setLoading(true)
   444→    try {
   445→      const params = new URLSearchParams()
   446→      if (search) params.set('search', search)
   447→      if (statusFilter !== 'all') params.set('status', statusFilter)
   448→      if (kycFilter !== 'all') params.set('kyc', kycFilter)
   449→      const res = await fetch(`/api/admin/users?${params}`)
   450→      const d = await res.json()
   451→      if (d.error) throw new Error(d.error)
   452→      setUsers(d.users || [])
   453→      setTotal(d.total || 0)
   454→    } catch (e: any) {
   455→      toast({ title: 'Failed to load users', description: e.message, variant: 'destructive' })
   456→    } finally {
   457→      setLoading(false)
   458→    }
   459→  }, [search, statusFilter, kycFilter, toast])
   460→
   461→  useEffect(() => { load() }, [load])
   462→
   463→  const [acting, setActing] = useState(false)
   464→  const act = async (userId: string, action: string, payload?: any) => {
   465→    if (acting) return // Prevent duplicate actions
   466→    setActing(true)
   467→    try {
   468→      const res = await fetch('/api/admin/users/action', {
   469→        method: 'POST',
   470→        headers: { 'Content-Type': 'application/json' },
   471→        body: JSON.stringify({ userId, action, ...payload }),
   472→      })
   473→      const d = await res.json()
