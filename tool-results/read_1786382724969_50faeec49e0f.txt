   310→      {/* Send dialog */}

   342→function AssetIcon({ asset }: { asset: string }) {

   348→    BNB: 'bg-yellow-500',
   349→    SOL: 'bg-purple-500',
   350→    XRP: 'bg-gray-500',
   351→    ADA: 'bg-blue-600',
   352→    DOGE: 'bg-yellow-400',
   353→    AVAX: 'bg-red-500',
   354→    LINK: 'bg-blue-500',

   358→  }

   366→function StatusBadge({ status }: { status: string }) {
   367→  const map: Record<string, { color: string; icon: any; label: string }> = {
   368→    COMPLETED: { color: 'text-green-500 bg-green-500/10', icon: CheckCircle2, label: 'Completed' },
   369→    PENDING: { color: 'text-yellow-500 bg-yellow-500/10', icon: Clock, label: 'Pending' },
   370→    FAILED: { color: 'text-red-500 bg-red-500/10', icon: XCircle, label: 'Failed' },
   371→    REJECTED: { color: 'text-red-500 bg-red-500/10', icon: XCircle, label: 'Rejected' },
   372→    CONFIRMED: { color: 'text-green-500 bg-green-500/10', icon: CheckCircle2, label: 'Confirmed' },
   373→  }
   374→  const config = map[status] || map.PENDING
   375→  const Icon = config.icon
   376→  return (
   377→    <span className={`inline-flex items-center gap-1 text-xs font-medium px-2 py-0.5 rounded ${config.color}`}>
   378→      <Icon className="h-3 w-3" />
   379→      {config.label}
   380→    </span>
   381→  )
   382→}

   384→function SendDialog({ wallet, wallets, onClose, onSuccess }: {
   385→  wallet: WalletData
   386→  wallets: WalletData[]
   387→  onClose: () => void
   388→  onSuccess: () => void
   389→}) {
   390→  const { toast } = useToast()
   391→  const [asset, setAsset] = useState(wallet.asset)
   392→  const [address, setAddress] = useState('')
   393→  const [network, setNetwork] = useState('')
   394→  const [amount, setAmount] = useState('')
   395→  const [loading, setLoading] = useState(false)
   396→
   397→  const NETWORKS: Record<string, string[]> = {
   398→    BTC: ['BTC'],
   399→    ETH: ['ERC20'],
   400→    USDT: ['TRC20', 'ERC20', 'BSC'],
   401→    USDC: ['ERC20'],
   402→    BNB: ['BSC'],
   403→    SOL: ['SOL'],
   404→    XRP: ['XRP'],
   405→    ADA: ['ADA'],
   406→    DOGE: ['DOGE'],
   407→    AVAX: ['AVAX'],
   408→    LINK: ['ERC20'],
   409→    DOT: ['DOT'],
   410→    MATIC: ['ERC20'],
   411→    LTC: ['LTC'],
   412→  }
   413→  const FEES: Record<string, Record<string, number>> = {
   414→    BTC: { BTC: 0.0001 },
   415→    ETH: { ERC20: 0.001 },
   416→    USDT: { TRC20: 1, ERC20: 5, BSC: 0.5 },
   417→    USDC: { ERC20: 5 },
   418→    BNB: { BSC: 0.001 },
   419→    SOL: { SOL: 0.01 },
   420→    XRP: { XRP: 0.1 },
   421→    ADA: { ADA: 0.2 },
   422→    DOGE: { DOGE: 5 },
   423→    AVAX: { AVAX: 0.01 },
   424→    LINK: { ERC20: 0.1 },
   425→    DOT: { DOT: 0.05 },
   426→    MATIC: { ERC20: 0.5 },
   427→    LTC: { LTC: 0.0005 },
   428→  }
   429→
   430→  const availableNetworks = NETWORKS[asset] || []
   431→  const selectedWallet = wallets.find(w => w.asset === asset)
   432→  const available = selectedWallet?.available ?? 0
   433→  const fee = FEES[asset]?.[network] ?? 0
   434→  const amountNum = parseFloat(amount) || 0
   435→  const total = amountNum + fee
   436→
   437→  useEffect(() => {
   438→    if (availableNetworks.length > 0 && !availableNetworks.includes(network)) {
   439→      setNetwork(availableNetworks[0])
   440→    }
   441→  }, [asset])
   442→
   443→  const submit = async () => {
   444→    if (!address) {
   445→      toast({ title: 'Enter destination address', variant: 'destructive' })
   446→      return
   447→    }
   448→    if (amountNum <= 0) {
   449→      toast({ title: 'Enter valid amount', variant: 'destructive' })
   450→      return
   451→    }
   452→    if (total > available) {
   453→      toast({ title: 'Insufficient balance', description: `Need ${total} ${asset}, have ${available}`, variant: 'destructive' })
   454→      return
   455→    }
   456→    setLoading(true)
   457→    try {
   458→      const res = await fetch('/api/wallet/send', {
   459→        method: 'POST',
   460→        headers: { 'Content-Type': 'application/json' },
   461→        body: JSON.stringify({ asset, network, address, amount: amountNum }),
   462→      })
   463→      const data = await res.json()
   464→      if (data.error) throw new Error(data.error)
   465→      toast({
   466→        title: 'Withdrawal submitted',
   467→        description: `${amountNum} ${asset} will be sent to ${shortAddr(address)} via ${network}`,
   468→      })
   469→      onSuccess()
   470→      onClose()
   471→    } catch (e: any) {
   472→      toast({ title: 'Withdrawal failed', description: e.message, variant: 'destructive' })
   473→    } finally {
   474→      setLoading(false)
   475→    }
   476→  }
   477→
   478→  return (
   479→    <Dialog open onOpenChange={onClose}>
   480→      <DialogContent className="max-w-md">
   481→        <DialogHeader>
   482→          <DialogTitle>Withdraw Crypto</DialogTitle>
   483→          <DialogDescription>Send crypto to an external wallet address</DialogDescription>
   484→        </DialogHeader>
   485→        <div className="space-y-3">
   486→          <div>
   487→            <label className="text-xs font-medium text-muted-foreground">Asset</label>
   488→            <Select value={asset} onValueChange={setAsset}>
   489→              <SelectTrigger><SelectValue /></SelectTrigger>
   490→              <SelectContent>
   491→                {wallets.filter(w => w.balance > 0 || ['BTC', 'ETH', 'USDT', 'USDC', 'BNB', 'SOL'].includes(w.asset)).map(w => (
   492→                  <SelectItem key={w.asset} value={w.asset}>
   493→                    {w.asset} - {w.assetName} ({formatQty(w.available)})
   494→                  </SelectItem>
   495→                ))}
   496→              </SelectContent>
   497→            </Select>
   498→          </div>
   499→
   500→          <div>
   501→            <label className="text-xs font-medium text-muted-foreground">Network</label>
   502→            <Select value={network} onValueChange={setNetwork}>

   572→function DepositDialog({ wallet, onClose, onSuccess }: {
   573→  wallet: WalletData
   574→  onClose: () => void
   575→  onSuccess: () => void
   576→}) {
   577→  const { toast } = useToast()
   578→  const [asset, setAsset] = useState(wallet.asset)
   579→  const [networks, setNetworks] = useState<any[]>([])
   580→  const [selectedNetwork, setSelectedNetwork] = useState('')
   581→  const [loading, setLoading] = useState(false)
   582→  const [copied, setCopied] = useState(false)
   583→
   584→  const loadAddress = async (a: string) => {
   585→    try {
   586→      const res = await fetch(`/api/wallet/address?asset=${a}`)
   587→      const data = await res.json()
   588→      setNetworks(data.networks || [])
   589→      if (data.networks?.length > 0) {
   590→        setSelectedNetwork(data.networks[0].network)
   591→      }
   592→    } catch {}
   593→  }
   594→
   595→  useEffect(() => { loadAddress(asset) }, [asset])
   596→  useEffect(() => { setAsset(wallet.asset) }, [wallet])
   597→
   598→  const current = networks.find(n => n.network === selectedNetwork)
   599→  const address = current?.address || wallet.depositAddress
   600→
   601→  const copy = async () => {
   602→    const ok = await copyToClipboard(address)
   603→    if (ok) {
   604→      setCopied(true)
   605→      toast({ title: 'Address copied' })
   606→      setTimeout(() => setCopied(false), 1500)
   607→    }
   608→  }
   609→
   610→  // Mock deposit (simulate receiving funds)
   611→  const simulateDeposit = async () => {
   612→    setLoading(true)
   613→    try {
   614→      const amount = asset === 'USDT' ? 1000 : asset === 'BTC' ? 0.05 : asset === 'ETH' ? 0.5 : 10
   615→      const res = await fetch('/api/wallet/deposit', {
   616→        method: 'POST',
   617→        headers: { 'Content-Type': 'application/json' },
   618→        body: JSON.stringify({ asset, network: selectedNetwork, amount }),
   619→      })
   620→      const data = await res.json()
   621→      if (data.error) throw new Error(data.error)
   622→      toast({
   623→        title: 'Deposit initiated',
   624→        description: `${amount} ${asset} will be credited after confirmations.`,
   625→      })
   626→      onSuccess()
   627→      onClose()
   628→    } catch (e: any) {
   629→      toast({ title: 'Deposit failed', description: e.message, variant: 'destructive' })
   630→    } finally {
   631→      setLoading(false)
   632→    }
   633→  }
   634→
   635→  return (
   636→    <Dialog open onOpenChange={onClose}>
   637→      <DialogContent className="max-w-md">
   638→        <DialogHeader>
   639→          <DialogTitle>Deposit Crypto</DialogTitle>
   640→          <DialogDescription>Receive crypto to your P2PEX wallet</DialogDescription>
   641→        </DialogHeader>
   642→        <div className="space-y-3">
   643→          <div>
   644→            <label className="text-xs font-medium text-muted-foreground">Asset</label>
   645→            <Select value={asset} onValueChange={setAsset}>
   646→              <SelectTrigger><SelectValue /></SelectTrigger>
   647→              <SelectContent>
   648→                {['BTC', 'ETH', 'USDT', 'USDC', 'BNB', 'SOL', 'XRP', 'ADA', 'DOGE', 'AVAX', 'LINK', 'DOT', 'MATIC', 'LTC'].map(a => (
   649→                  <SelectItem key={a} value={a}>{a}</SelectItem>
   650→                ))}
   651→              </SelectContent>
   652→            </Select>
   653→          </div>
   654→
   655→          <div>
   656→            <label className="text-xs font-medium text-muted-foreground">Network</label>
   657→            <Select value={selectedNetwork} onValueChange={setSelectedNetwork}>
   658→              <SelectTrigger><SelectValue placeholder="Select network" /></SelectTrigger>
   659→              <SelectContent>
   660→                {networks.map(n => (
   661→                  <SelectItem key={n.network} value={n.network}>
   662→                    {n.network} {n.fee ? `(fee: ${n.fee})` : ''}
   663→                  </SelectItem>
   664→                ))}
   665→              </SelectContent>
   666→            </Select>
   667→            {networks.length > 1 && (
   668→              <p className="text-xs text-yellow-600 dark:text-yellow-500 mt-1">
   669→                ⚠ Make sure to send via {selectedNetwork}. Other networks may not be credited.
   670→              </p>
   671→            )}
   672→          </div>
   673→
   674→          {/* QR + Address */}
   675→          <div className="flex flex-col items-center py-4 bg-muted/30 rounded-lg">
   676→            <div className="bg-white p-3 rounded-lg mb-3">
   677→              <QrCode className="h-32 w-32 text-black" />
   678→            </div>
   679→            <p className="text-xs text-muted-foreground mb-1">Deposit Address ({selectedNetwork})</p>
   680→            <p className="text-xs font-mono break-all px-4 text-center max-w-xs">{address}</p>
   681→            <Button variant="outline" size="sm" className="mt-2" onClick={copy}>
   682→              {copied ? <CheckCircle2 className="h-3 w-3 mr-1" /> : <Copy className="h-3 w-3 mr-1" />}
   683→              {copied ? 'Copied!' : 'Copy Address'}
   684→            </Button>
   685→          </div>
   686→
   687→          <div className="text-xs text-muted-foreground bg-yellow-500/10 p-2 rounded">
   688→            <p className="font-medium text-yellow-600 dark:text-yellow-500 mb-1">⚠ Important</p>
   689→            <ul className="list-disc list-inside space-y-0.5">
   690→              <li>Send only {asset} to this address</li>

   693→            </ul>
   694→          </div>

   696→          {/* Demo deposit button */}

   698→            variant="default"
   699→            className="w-full"
   700→            onClick={simulateDeposit}
   701→            disabled={loading}
   702→          >

   709→      </DialogContent>
   710→    </Dialog>

   718→function TransferDialog({ wallet, wallets, onClose, onSuccess }: {
   719→  wallet: WalletData
   720→  wallets: WalletData[]
   721→  onClose: () => void
   722→  onSuccess: () => void
   723→}) {
   724→  const { toast } = useToast()
   725→  const [asset, setAsset] = useState(wallet.asset)
   726→  const [recipientEmail, setRecipientEmail] = useState('')
   727→  const [amount, setAmount] = useState('')
   728→  const [note, setNote] = useState('')
   729→  const [loading, setLoading] = useState(false)
   730→
   731→  const selectedWallet = wallets.find(w => w.asset === asset)
   732→  const available = selectedWallet?.available ?? 0
   733→  const amountNum = parseFloat(amount) || 0
   734→
   735→  const submit = async () => {
   736→    if (!recipientEmail) {
   737→      toast({ title: 'Enter recipient email', variant: 'destructive' })
   738→      return
   739→    }
   740→    if (amountNum <= 0) {
   741→      toast({ title: 'Enter valid amount', variant: 'destructive' })
   742→      return
   743→    }
   744→    if (amountNum > available) {
   745→      toast({ title: 'Insufficient balance', description: `Available: ${available} ${asset}`, variant: 'destructive' })
   746→      return
   747→    }
   748→    setLoading(true)
   749→    try {
   750→      const res = await fetch('/api/wallet/transfer', {
   751→        method: 'POST',
   752→        headers: { 'Content-Type': 'application/json' },
   753→        body: JSON.stringify({ asset, amount: amountNum, recipientEmail, note }),
   754→      })
   755→      const data = await res.json()
   756→      if (data.error) throw new Error(data.error)
   757→      toast({
   758→        title: 'Transfer successful',
   759→        description: `${amountNum} ${asset} sent to ${recipientEmail}`,
   760→      })
   761→      onSuccess()
   762→      onClose()
   763→    } catch (e: any) {
   764→      toast({ title: 'Transfer failed', description: e.message, variant: 'destructive' })
   765→    } finally {
   766→      setLoading(false)
   767→    }

   776→            Internal Transfer
   777→          </DialogTitle>
   778→          <DialogDescription>
   779→            Send crypto to another P2PEX user instantly — no blockchain fees, no confirmation delays.
   780→          </DialogDescription>
   781→        </DialogHeader>
   782→
   783→        <div className="space-y-3">
   784→          {/* Info banner */}
   785→          <div className="bg-blue-500/10 border border-blue-500/30 rounded-lg p-3 text-xs">
   786→            <div className="flex items-start gap-2">
   787→              <Zap className="h-4 w-4 text-blue-500 mt-0.5 flex-shrink-0" />

   789→                <p className="font-medium mb-0.5">Internal transfers are:</p>
   790→                <ul className="list-disc list-inside space-y-0.5 text-muted-foreground">
   791→                  <li>Free — no network fees</li>
   792→                  <li>Instant — confirmed immediately</li>
   793→                  <li>Only between P2PEX users (by email)</li>
   794→                </ul>
   795→              </div>
   796→            </div>
   797→          </div>

   799→          <div>
   800→            <label className="text-xs font-medium text-muted-foreground">Asset</label>

   802→              <SelectTrigger><SelectValue /></SelectTrigger>
