    16→// Local order book component (live from socket)
    17→function OrderBook({ symbol }: { symbol: string }) {
    18→  const { orderBook, ticker } = useMarketSocket(symbol)
    19→  const [showDepth, setShowDepth] = useState(false)
    20→
    21→  const maxBidQty = useMemo(() => Math.max(...orderBook.bids.map(b => b[1]), 0), [orderBook.bids])
    22→  const maxAskQty = useMemo(() => Math.max(...orderBook.asks.map(a => a[1]), 0), [orderBook.asks])
    23→  const maxQty = Math.max(maxBidQty, maxAskQty, 1)
    24→
    25→  const isUp = (ticker?.changePercent ?? 0) >= 0
    26→
    27→  return (
    28→    <div className="flex flex-col h-full">
    29→      <div className="flex items-center justify-between px-3 py-2 border-b border-border">
    30→        <span className="text-xs font-medium">Order Book</span>
    31→        <Button
    32→          variant="ghost"
    33→          size="sm"
    34→          className="h-6 text-xs px-2"
    35→          onClick={() => setShowDepth(!showDepth)}
    36→        >
    37→          {showDepth ? 'List' : 'Depth'}
    38→        </Button>
    39→      </div>
    40→
    41→      {showDepth ? (
    42→        <div className="p-2">
    43→          <DepthChart
    44→            bids={orderBook.bids}
    45→            asks={orderBook.asks}
    46→            lastPrice={ticker?.lastPrice}
    47→            height={280}
    48→          />
    49→        </div>
    50→      ) : (
    51→        <div className="flex-1 flex flex-col text-xs overflow-hidden">
    52→          {/* Header */}
    53→          <div className="grid grid-cols-3 gap-2 px-3 py-1.5 text-muted-foreground border-b border-border">
    54→            <span className="text-left">Price</span>
    55→            <span className="text-right">Amount</span>
    56→            <span className="text-right">Total</span>
    57→          </div>
    58→
    59→          {/* Asks (reversed: highest at top, lowest at bottom near spread) */}
    60→          <div className="flex-1 overflow-y-auto min-h-0 max-h-[180px]">
    61→            {[...orderBook.asks].slice(0, 12).reverse().map(([price, qty], i) => (
    62→              <div
    63→                key={`a-${i}`}
    64→                className="grid grid-cols-3 gap-2 px-3 py-0.5 tabular-nums relative hover:bg-muted/30"
    65→              >
    66→                <div
    67→                  className="absolute right-0 top-0 bottom-0 bg-red-500/10"
    68→                  style={{ width: `${(qty / maxQty) * 100}%` }}
    69→                />
    70→                <span className="text-red-500 relative z-10">{formatPrice(price)}</span>
    71→                <span className="text-right relative z-10">{formatQty(qty)}</span>
    72→                <span className="text-right text-muted-foreground relative z-10">{formatQty(price * qty)}</span>
    73→              </div>
    74→            ))}
    75→          </div>
    76→
    77→          {/* Spread / Last price */}
    78→          <div className="border-y border-border px-3 py-2 bg-muted/30">
    79→            <div className="flex items-center justify-between">
    80→              <span className={`text-base font-bold ${isUp ? 'text-green-500' : 'text-red-500'}`}>
    81→                {formatPrice(ticker?.lastPrice ?? 0)}
    82→              </span>
    83→              <span className="text-xs text-muted-foreground">
    84→                {isUp ? <ArrowUp className="inline h-3 w-3" /> : <ArrowDown className="inline h-3 w-3" />}
    85→                {formatPercent(ticker?.changePercent ?? 0)}

    87→            </div>
    88→          </div>
    89→
    90→          {/* Bids */}
    91→          <div className="flex-1 overflow-y-auto min-h-0 max-h-[180px]">
    92→            {orderBook.bids.slice(0, 12).map(([price, qty], i) => (
    93→              <div
    94→                key={`b-${i}`}
    95→                className="grid grid-cols-3 gap-2 px-3 py-0.5 tabular-nums relative hover:bg-muted/30"
    96→              >
    97→                <div
    98→                  className="absolute right-0 top-0 bottom-0 bg-green-500/10"
    99→                  style={{ width: `${(qty / maxQty) * 100}%` }}
   100→                />
   101→                <span className="text-green-500 relative z-10">{formatPrice(price)}</span>
   102→                <span className="text-right relative z-10">{formatQty(qty)}</span>
   103→                <span className="text-right text-muted-foreground relative z-10">{formatQty(price * qty)}</span>
   104→              </div>
   105→            ))}
   106→          </div>
   107→        </div>

   109→    </div>
   110→  )

   146→// Trading panel - place orders
   147→function TradePanel({ symbol }: { symbol: string }) {
   148→  const { ticker } = useMarketSocket(symbol)
   149→  const { user } = useAppStore()
   150→  const { toast } = useToast()
   151→  const [side, setSide] = useState<'BUY' | 'SELL'>('BUY')
   152→  const [orderType, setOrderType] = useState<'LIMIT' | 'MARKET'>('LIMIT')
   153→  const [price, setPrice] = useState('')
   154→  const [qty, setQty] = useState('')
   155→  const [pct, setPct] = useState<number>(0)
   156→  const [loading, setLoading] = useState(false)
   157→
   158→  // Wallets for current pair
   159→  const [balances, setBalances] = useState<{ base: number; quote: number } | null>(null)
   160→  const base = symbol.replace(/(USDT|USDC|BTC|ETH|BNB)$/, '')
   161→  const quote = symbol.startsWith(base) ? symbol.slice(base.length) : 'USDT'
   162→
   163→  const loadBalances = useCallback(async () => {
   164→    if (!user) return
   165→    try {
   166→      const res = await fetch('/api/wallet')
   167→      const data = await res.json()
   168→      if (data.wallets) {
   169→        const baseWallet = data.wallets.find((w: any) => w.asset === base)
   170→        const quoteWallet = data.wallets.find((w: any) => w.asset === quote)
   171→        setBalances({
   172→          base: baseWallet?.available ?? 0,
   173→          quote: quoteWallet?.available ?? 0,
   174→        })
   175→      }
   176→    } catch {}
   177→  }, [user, base, quote])
   178→
   179→  useEffect(() => {
   180→    loadBalances()
   181→    const t = setInterval(loadBalances, 5000)
   182→    return () => clearInterval(t)
   183→  }, [loadBalances])
   184→
   185→  // Update price when ticker changes (only if user hasn't manually edited)
   186→  const [userEditedPrice, setUserEditedPrice] = useState(false)
   187→  useEffect(() => {
   188→    if (!userEditedPrice && ticker && orderType === 'LIMIT') {
   189→      setPrice(formatPrice(ticker.lastPrice, { maxDigits: 6 }))
   190→    }
   191→  }, [ticker?.lastPrice, orderType, userEditedPrice])
   192→
   193→  // Reset price edit flag when switching symbol
   194→  useEffect(() => {
   195→    setUserEditedPrice(false)
   196→    setQty('')
   197→    setPct(0)
   198→  }, [symbol])
   199→
   200→  const lastPrice = ticker?.lastPrice ?? 0
   201→  const currentPrice = parseFloat(price) || lastPrice
   202→  const currentQty = parseFloat(qty) || 0
   203→  const total = currentPrice * currentQty
   204→
   205→  // Apply percentage
   206→  const applyPct = (p: number) => {
   207→    setPct(p)
   208→    if (!balances) return
   209→    if (side === 'BUY') {
   210→      const spendable = balances.quote
   211→      const spend = (spendable * p) / 100
   212→      const newQty = currentPrice > 0 ? spend / currentPrice : 0
   213→      setQty(newQty > 0 ? newQty.toFixed(6) : '')
   214→    } else {
   215→      const newQty = (balances.base * p) / 100
   216→      setQty(newQty > 0 ? newQty.toFixed(6) : '')
   217→    }
   218→  }
   219→
   220→  const submit = async () => {
   221→    if (!user) {
   222→      toast({ title: 'Please log in to trade', variant: 'destructive' })
   223→      return
   224→    }
   225→    if (orderType === 'LIMIT' && (!price || parseFloat(price) <= 0)) {
   226→      toast({ title: 'Enter a valid price', variant: 'destructive' })
   227→      return
   228→    }
   229→    if (!qty || parseFloat(qty) <= 0) {
   230→      toast({ title: 'Enter a valid quantity', variant: 'destructive' })
   231→      return
   232→    }
   233→    setLoading(true)
   234→    try {
   235→      const res = await fetch('/api/trade/order', {
   236→        method: 'POST',
   237→        headers: { 'Content-Type': 'application/json' },
   238→        body: JSON.stringify({
   239→          symbol,
   240→          side,
   241→          type: orderType,
   242→          price: orderType === 'LIMIT' ? parseFloat(price) : currentPrice,
   243→          quantity: parseFloat(qty),
   244→        }),
   245→      })
   246→      const data = await res.json()
   247→      if (data.error) throw new Error(data.error)
   248→
   249→      toast({
   250→        title: 'Order placed',
   251→        description: `${side} ${formatQty(parseFloat(qty))} ${base} @ ${orderType === 'MARKET' ? 'market' : formatPrice(parseFloat(price))}`,
   252→      })
   253→      setQty('')
   254→      setPct(0)
   255→      loadBalances()
   256→    } catch (e: any) {
   257→      toast({ title: 'Order failed', description: e.message, variant: 'destructive' })
   258→    } finally {
   259→      setLoading(false)
   260→    }
   261→  }
   262→
   263→  if (!user) {
   264→    return (
   265→      <div className="p-6 text-center">
   266→        <p className="text-sm text-muted-foreground mb-3">Log in to start trading</p>
   267→        <Button variant="default" disabled>Place Order</Button>
   268→      </div>
   269→    )
   270→  }
   271→
   272→  return (
   273→    <div className="flex flex-col">
   274→      {/* Buy/Sell tabs */}
   275→      <div className="grid grid-cols-2 gap-1 p-2">
   276→        <Button
   277→          variant={side === 'BUY' ? 'default' : 'outline'}
   278→          className={side === 'BUY' ? 'bg-green-500 hover:bg-green-600 text-white' : ''}
   279→          onClick={() => setSide('BUY')}
   280→        >
   281→          Buy {base}
   282→        </Button>
   283→        <Button
   284→          variant={side === 'SELL' ? 'default' : 'outline'}
   285→          className={side === 'SELL' ? 'bg-red-500 hover:bg-red-600 text-white' : ''}
   286→          onClick={() => setSide('SELL')}
   287→        >
   288→          Sell {base}
   289→        </Button>
   290→      </div>
   291→
   292→      <div className="px-3 pb-2 flex gap-1">
   293→        <Button
   294→          variant={orderType === 'LIMIT' ? 'secondary' : 'ghost'}
   295→          size="sm"
   296→          className="text-xs flex-1"
   297→          onClick={() => setOrderType('LIMIT')}
   298→        >
   299→          Limit
   300→        </Button>
   301→        <Button
   302→          variant={orderType === 'MARKET' ? 'secondary' : 'ghost'}
   303→          size="sm"
   304→          className="text-xs flex-1"
   305→          onClick={() => setOrderType('MARKET')}
   306→        >
   307→          Market
   308→        </Button>
   309→      </div>
   310→
   311→      <div className="px-3 pb-3 space-y-2">
   312→        {/* Available balance */}
   313→        <div className="flex justify-between text-xs">
   314→          <span className="text-muted-foreground">Available</span>
   315→          <span className="tabular-nums">
   316→            {side === 'BUY'
   317→              ? `${formatQty(balances?.quote ?? 0)} ${quote}`
   318→              : `${formatQty(balances?.base ?? 0)} ${base}`}
   319→          </span>
   320→        </div>
   321→
   322→        {/* Price input (only for LIMIT) */}
   323→        {orderType === 'LIMIT' && (
   324→          <div>
   325→            <label className="text-xs text-muted-foreground">Price ({quote})</label>
   326→            <div className="relative">
   327→              <Input
   328→                type="number"
   329→                value={price}
   330→                onChange={e => { setPrice(e.target.value); setUserEditedPrice(true) }}
   331→                placeholder="0.00"
   332→                className="pr-12 tabular-nums"
   333→              />
   334→              <span className="absolute right-3 top-1/2 -translate-y-1/2 text-xs text-muted-foreground">{quote}</span>
   335→            </div>
   336→          </div>
   337→        )}
   338→
   339→        {/* Quantity */}
   340→        <div>
   341→          <label className="text-xs text-muted-foreground">Amount ({base})</label>
   342→          <div className="relative">
   343→            <Input
   344→              type="number"
   345→              value={qty}
   346→              onChange={e => { setQty(e.target.value); setPct(0) }}
   347→              placeholder="0.00"
   348→              className="pr-12 tabular-nums"
   349→            />
   350→            <span className="absolute right-3 top-1/2 -translate-y-1/2 text-xs text-muted-foreground">{base}</span>
   351→          </div>
   352→        </div>
   353→
   354→        {/* Percentage slider */}
   355→        <div className="grid grid-cols-4 gap-1">
   356→          {[25, 50, 75, 100].map(p => (
   357→            <Button
   358→              key={p}
   359→              variant={pct === p ? 'default' : 'outline'}
   360→              size="sm"
   361→              className="text-xs h-7"
   362→              onClick={() => applyPct(p)}
   363→            >
   364→              {p}%
   365→            </Button>
   366→          ))}
   367→        </div>
   368→
   369→        {/* Total */}
   370→        <div>
   371→          <label className="text-xs text-muted-foreground">Total ({quote})</label>
   372→          <Input
   373→            type="text"
   374→            value={formatQty(total)}
   375→            readOnly
   376→            className="tabular-nums bg-muted/30"
   377→          />
   378→        </div>
   379→
   380→        {/* Submit */}
   381→        <Button
   382→          className={`w-full mt-2 ${
   383→            side === 'BUY'
   384→              ? 'bg-green-500 hover:bg-green-600 text-white'
   385→              : 'bg-red-500 hover:bg-red-600 text-white'
   386→          }`}
   387→          onClick={submit}
   388→          disabled={loading}
   389→        >
   390→          {loading ? 'Placing...' : `${side} ${base}`}
   391→        </Button>
   392→      </div>
   393→    </div>
   394→  )
   395→}

   397→// User's open orders
   398→function OpenOrders({ user }: { user: any }) {
   399→  const { toast } = useToast()
   400→  const [orders, setOrders] = useState<any[]>([])
   401→  const [loading, setLoading] = useState(true)
   402→
   403→  const load = useCallback(async () => {
   404→    if (!user) { setOrders([]); setLoading(false); return }
   405→    try {
   406→      const res = await fetch('/api/trade/order?status=OPEN')
   407→      const data = await res.json()
   408→      setOrders(data.orders || [])
   409→    } catch {
   410→      setOrders([])
   411→    } finally {
   412→      setLoading(false)
   413→    }
   414→  }, [user])
   415→
   416→  useEffect(() => {
   417→    load()
   418→    const t = setInterval(load, 5000)
   419→    return () => clearInterval(t)
   420→  }, [load])
   421→
   422→  const cancel = async (orderId: string) => {
   423→    try {
   424→      await fetch('/api/trade/cancel', {
   425→        method: 'POST',
   426→        headers: { 'Content-Type': 'application/json' },
   427→        body: JSON.stringify({ orderId }),
   428→      })
   429→      toast({ title: 'Order canceled' })
   430→      load()
   431→    } catch (e: any) {
   432→      toast({ title: 'Cancel failed', description: e.message, variant: 'destructive' })
   433→    }
   434→  }
   435→
   436→  if (!user) return null
   437→
   438→  return (
   439→    <div className="rounded-lg border border-border bg-card overflow-hidden">
   440→      <div className="px-3 py-2 border-b border-border">
   441→        <span className="text-xs font-medium">Open Orders ({orders.length})</span>
   442→      </div>
   443→      <div className="overflow-x-auto">
   444→        <table className="w-full text-xs">
   445→          <thead className="bg-muted/30 text-muted-foreground">
   446→            <tr>
   447→              <th className="px-3 py-2 text-left font-medium">Pair</th>
   448→              <th className="px-3 py-2 text-left font-medium">Type</th>
   449→              <th className="px-3 py-2 text-right font-medium">Price</th>
   450→              <th className="px-3 py-2 text-right font-medium">Amount</th>
   451→              <th className="px-3 py-2 text-right font-medium">Filled</th>
   452→              <th className="px-3 py-2 text-right font-medium">Action</th>
   453→            </tr>
   454→          </thead>
   455→          <tbody>
   456→            {loading ? (
   457→              <tr><td colSpan={6} className="px-3 py-6 text-center text-muted-foreground">Loading...</td></tr>
   458→            ) : orders.length === 0 ? (
   459→              <tr><td colSpan={6} className="px-3 py-6 text-center text-muted-foreground">No open orders</td></tr>
   460→            ) : (
   461→              orders.map(o => (
   462→                <tr key={o.id} className="border-t border-border/40">
   463→                  <td className="px-3 py-2">
   464→                    <span className={o.side === 'BUY' ? 'text-green-500' : 'text-red-500'}>
   465→                      {o.side === 'BUY' ? 'Buy' : 'Sell'}
   466→                    </span>{' '}
   467→                    {o.symbol}
   468→                  </td>
   469→                  <td className="px-3 py-2 text-muted-foreground">{o.type}</td>
   470→                  <td className="px-3 py-2 text-right tabular-nums">{formatPrice(o.price)}</td>
   471→                  <td className="px-3 py-2 text-right tabular-nums">{formatQty(o.quantity)}</td>
   472→                  <td className="px-3 py-2 text-right tabular-nums">{((o.filledQty / o.quantity) * 100).toFixed(1)}%</td>
   473→                  <td className="px-3 py-2 text-right">
   474→                    <Button
   475→                      variant="ghost"
   476→                      size="sm"
   477→                      className="h-6 text-xs text-red-500"
   478→                      onClick={() => cancel(o.id)}
   479→                    >
   480→                      <X className="h-3 w-3" /> Cancel
   481→                    </Button>
   482→                  </td>
   483→                </tr>
   484→              ))
   485→            )}
   486→          </tbody>
   487→        </table>
   488→      </div>
   489→    </div>
   490→  )
   491→}

   941→// =================== PROFIT/LOSS PANEL ===================
   942→function ProfitLossPanel({ symbol, user }: { symbol: string; user: any }) {
   943→  const [trades, setTrades] = useState<any[]>([])
   944→  const [loading, setLoading] = useState(true)
   945→
   946→  const load = useCallback(async () => {
   947→    if (!user) { setTrades([]); setLoading(false); return }
   948→    try {
   949→      const res = await fetch(`/api/trade/trades?symbol=${symbol}&limit=50`)
   950→      const data = await res.json()
   951→      setTrades(data.trades || [])
   952→    } catch {
   953→      setTrades([])
   954→    } finally {
   955→      setLoading(false)
   956→    }
   957→  }, [user, symbol])
   958→
   959→  useEffect(() => {
   960→    load()
   961→    const t = setInterval(load, 10000)
   962→    return () => clearInterval(t)
   963→  }, [load])
   964→
   965→  const base = symbol.replace(/(USDT|USDC|BTC|ETH|BNB)$/, '')
   966→  const quote = symbol.startsWith(base) ? symbol.slice(base.length) : 'USDT'
   967→
   968→  // Calculate PnL from trade history
   969→  // For BUY trades: profit = (currentPrice - buyPrice) * quantity
   970→  // For SELL trades: profit = (sellPrice - avgBuyPrice) * quantity
   971→  const prices: Record<string, number> = {
   972→    BTC: 67500, ETH: 3450, BNB: 585, SOL: 165, XRP: 0.62, ADA: 0.45,
   973→    DOGE: 0.16, AVAX: 38, LINK: 18.5, DOT: 7.2, MATIC: 0.72, LTC: 85,
   974→    USDT: 1, USDC: 1,
   975→  }
   976→  const currentPrice = prices[base] || 0
   977→
   978→  // Calculate total PnL
   979→  let totalBuyCost = 0
   980→  let totalBuyQty = 0
   981→  let totalSellRevenue = 0
   982→  let totalSellQty = 0
   983→  let realizedPnL = 0
   984→
   985→  // First pass: calculate average buy price
   986→  const buyTrades = trades.filter(t => t.side === 'BUY')
   987→  const sellTrades = trades.filter(t => t.side === 'SELL')
   988→
   989→  for (const t of buyTrades) {
   990→    totalBuyCost += t.price * t.quantity
   991→    totalBuyQty += t.quantity
   992→  }
   993→  const avgBuyPrice = totalBuyQty > 0 ? totalBuyCost / totalBuyQty : 0
   994→
   995→  for (const t of sellTrades) {
   996→    totalSellRevenue += t.price * t.quantity
   997→    totalSellQty += t.quantity
   998→    // Realized PnL = (sell price - avg buy price) * sell qty
   999→    realizedPnL += (t.price - avgBuyPrice) * t.quantity
  1000→  }
  1001→
  1002→  // Unrealized PnL = (current price - avg buy price) * remaining qty
  1003→  const remainingQty = totalBuyQty - totalSellQty
  1004→  const unrealizedPnL = remainingQty > 0 ? (currentPrice - avgBuyPrice) * remainingQty : 0
  1005→
  1006→  // Total PnL = realized + unrealized
  1007→  const totalPnL = realizedPnL + unrealizedPnL
  1008→  const pnlPercent = totalBuyCost > 0 ? (totalPnL / totalBuyCost) * 100 : 0
  1009→
  1010→  const isProfit = totalPnL >= 0
  1011→
  1012→  if (!user) return null
  1013→
  1014→  return (
