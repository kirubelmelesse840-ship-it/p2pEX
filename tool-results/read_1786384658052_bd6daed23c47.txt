   116→        // Detect status changes and notify the user
   117→        for (const o of newOrders) {
   118→          const prevStatus = prevOrderStatuses[o.id]
   119→          if (prevStatus && prevStatus !== o.status) {
   120→            // Status changed — show notification
   121→            if (o.status === 'PAID' && o.myRole === 'BUYER') {
   122→              toast({ title: '✅ Payment Approved!', description: `Your payment for ${o.amount} ${o.asset} has been verified. The seller will release your crypto shortly.` })

   124→              toast({ title: '💰 Payment Verified', description: `The buyer's payment for ${o.amount} ${o.asset} has been verified. You can now release the crypto.` })

   126→              toast({ title: '❌ Payment Rejected', description: `Your payment for ${o.amount} ${o.asset} was rejected. The order has been canceled. You can place a new order.`, variant: 'destructive' })

   128→              toast({ title: '❌ Order Canceled', description: `The order for ${o.amount} ${o.asset} has been canceled.`, variant: 'destructive' })

   130→              toast({ title: '🎉 Trade Completed!', description: `Your trade of ${o.amount} ${o.asset} has been completed successfully.` })
   131→            }
   132→          }
   133→        }

   194→      <div className="flex items-center justify-between mb-4 mt-1">

   202→        {user && (
   203→          <div className="flex items-center gap-2">
   204→            <Button variant="outline" size="sm" onClick={() => { load() }} className="gap-1.5 cursor-pointer hover:bg-primary/10">

   338→function ListingCard({ listing, onTrade }: { listing: Listing; onTrade: () => void }) {
   339→  const { user } = useAppStore()
   340→  const isMine = user && listing.user.name === user.name
   341→  const hasVerification = listing.user.kycVerified
   342→  const kycLevel = listing.user.kycLevel || 0
   343→  // Use real trades count and rating from the listing (set by admin)
   344→  const tradesCount = listing.tradesCount || 0

   347→  const firstMethod = listing.paymentMethods[0]
   348→  const paymentDetail = listing.paymentDetails?.[firstMethod]
   349→  const advertiserName = paymentDetail?.name || listing.user.name

   355→        <div className="flex items-center gap-2">

   357→            {advertiserName.slice(0, 2).toUpperCase()}

   424→          <Button
   425→            variant={listing.side === 'SELL' ? 'default' : 'outline'}

   435→      {/* Available amount */}

   770→                  )}
   771→                </button>

   773→            </div>

   778→            <div className="bg-yellow-500/10 border border-yellow-500/30 rounded p-2 text-xs">

   780→              <p className="text-muted-foreground">{listing.terms}</p>

   792→            </Button>
   793→          </DialogFooter>

   795→      </DialogContent>
