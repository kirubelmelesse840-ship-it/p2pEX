   291→function MetricCard({ icon, label, value, sub, color }: {

   320→function UsersTab() {
   321→  const { toast } = useToast()
   322→  const [users, setUsers] = useState<any[]>([])
   323→  const [total, setTotal] = useState(0)
   324→  const [loading, setLoading] = useState(true)
   325→  const [search, setSearch] = useState('')
   326→  const [statusFilter, setStatusFilter] = useState('all')
   327→  const [kycFilter, setKycFilter] = useState('all')
   328→  const [walletDialog, setWalletDialog] = useState<any>(null)
   329→  const [banDialog, setBanDialog] = useState<any>(null)
   330→  const [docDialog, setDocDialog] = useState<any>(null)
   331→
   332→  const load = useCallback(async () => {
   333→    setLoading(true)
   334→    try {
   335→      const params = new URLSearchParams()
   336→      if (search) params.set('search', search)
   337→      if (statusFilter !== 'all') params.set('status', statusFilter)
   338→      if (kycFilter !== 'all') params.set('kyc', kycFilter)
   339→      const res = await fetch(`/api/admin/users?${params}`)
   340→      const d = await res.json()
   341→      if (d.error) throw new Error(d.error)
   342→      setUsers(d.users || [])
   343→      setTotal(d.total || 0)
   344→    } catch (e: any) {
   345→      toast({ title: 'Failed to load users', description: e.message, variant: 'destructive' })
   346→    } finally {
   347→      setLoading(false)
   348→    }
   349→  }, [search, statusFilter, kycFilter, toast])
   350→
   351→  useEffect(() => { load() }, [load])
   352→
   353→  const act = async (userId: string, action: string, payload?: any) => {
   354→    try {
   355→      const res = await fetch('/api/admin/users/action', {
   356→        method: 'POST',
   357→        headers: { 'Content-Type': 'application/json' },
   358→        body: JSON.stringify({ userId, action, ...payload }),
   359→      })
   360→      const d = await res.json()
   361→      if (d.error) throw new Error(d.error)
   362→      toast({ title: 'Action completed', description: d.message })
   363→      load()
   364→    } catch (e: any) {
   365→      toast({ title: 'Action failed', description: e.message, variant: 'destructive' })
   366→    }
   367→  }
   368→
   369→  return (
   370→    <div className="space-y-3">
   371→      {/* Filters */}
   372→      <div className="flex flex-wrap items-center gap-2 p-3 bg-card rounded-lg border border-border">
   373→        <div className="relative flex-1 min-w-[200px]">
   374→          <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   375→          <Input
   376→            placeholder="Search by name or email..."
   377→            value={search}
   378→            onChange={e => setSearch(e.target.value)}
   379→            className="pl-8 h-9"
   380→          />
   381→        </div>
   382→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   383→          <SelectTrigger className="w-32 h-9"><SelectValue /></SelectTrigger>
   384→          <SelectContent>
   385→            <SelectItem value="all">All Status</SelectItem>
   386→            <SelectItem value="active">Active</SelectItem>
   387→            <SelectItem value="banned">Banned</SelectItem>
   388→          </SelectContent>
   389→        </Select>
   390→        <Select value={kycFilter} onValueChange={setKycFilter}>
   391→          <SelectTrigger className="w-32 h-9"><SelectValue /></SelectTrigger>
   392→          <SelectContent>
   393→            <SelectItem value="all">All KYC</SelectItem>
   394→            <SelectItem value="pending">⏳ Pending Review</SelectItem>
   395→            <SelectItem value="verified">Verified</SelectItem>
   396→            <SelectItem value="unverified">Unverified</SelectItem>
   397→            <SelectItem value="level1">Level 1</SelectItem>
   398→            <SelectItem value="level2">Level 2</SelectItem>
   399→          </SelectContent>
   400→        </Select>
   401→        <Button variant="outline" size="sm" onClick={load}>
   402→          <RefreshCw className="h-3.5 w-3.5" /> Refresh
   403→        </Button>
   404→      </div>
   405→
   406→      <div className="text-xs text-muted-foreground">Showing {users.length} of {total} users</div>
   407→
   408→      {/* Users table */}
   409→      <div className="bg-card border border-border rounded-lg overflow-hidden">
   410→        <div className="overflow-x-auto">
   411→          <table className="w-full text-sm">
   412→            <thead className="bg-muted/40 text-xs text-muted-foreground">
   413→              <tr>
   414→                <th className="px-3 py-2 text-left font-medium">User</th>
   415→                <th className="px-3 py-2 text-left font-medium hidden md:table-cell">Email</th>
   416→                <th className="px-3 py-2 text-left font-medium">KYC</th>
   417→                <th className="px-3 py-2 text-left font-medium">Status</th>
   418→                <th className="px-3 py-2 text-right font-medium hidden sm:table-cell">Balance</th>
   419→                <th className="px-3 py-2 text-right font-medium">Actions</th>

   478→                        {u.kycStatus === 'APPROVED' && (

   544→    </div>
