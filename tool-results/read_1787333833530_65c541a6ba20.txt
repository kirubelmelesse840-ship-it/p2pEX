   250→function DashboardTab() {
   251→  const [data, setData] = useState<any>(null)
   252→  const [loading, setLoading] = useState(false)
   253→  const [refreshing, setRefreshing] = useState(false)
   254→  const [error, setError] = useState<string | null>(null)
   255→
   256→  const load = useCallback(async () => {
   257→    // Manual refresh button should feel instant — show cached data immediately, then update
   258→    const cached = sessionStorage.getItem('admin-stats')
   259→    if (cached && !data) {
   260→      try { setData(JSON.parse(cached)) } catch {}
   261→    }
   262→    try {
   263→      const res = await fetch('/api/admin/stats')
   264→      // Handle empty response body (happens when function cold-starts or times out)
   265→      const text = await res.text()
   266→      if (!text) {
   267→        // Empty response — keep existing data, don't overwrite with an error
   268→        if (!data) setError('Server returned empty response. Tap Refresh to try again.')
   269→        return
   270→      }
   271→      const d = JSON.parse(text)
   272→      if (d.error) throw new Error(d.error)
   273→      setData(d)
   274→      setError(null)
   275→      sessionStorage.setItem('admin-stats', JSON.stringify(d))
   276→    } catch (e: any) {
   277→      console.error('[admin dashboard]', e)
   278→      // Only set error if we don't have any data yet — otherwise keep showing the cached data
   279→      if (!data) setError(e.message || 'Failed to load dashboard')
   280→    } finally {
   281→      setLoading(false)
   282→      setRefreshing(false)
   283→    }
   284→  }, [data])
   285→
   286→  useEffect(() => {
   287→    load() // Load once on mount — no auto-refresh (admin taps Refresh button manually)
   288→  }, [load])
   289→
   290→  if (!data) {
   291→    return <div className="text-center py-12 text-muted-foreground">Loading dashboard...</div>
   292→  }
   293→
   294→  if (error && !data) {
   295→    return (
   296→      <div className="text-center py-12">
   297→        <AlertTriangle className="h-12 w-12 mx-auto text-red-500 mb-3" />
   298→        <p className="text-sm text-muted-foreground mb-2">Failed to load dashboard</p>
   299→        <p className="text-xs text-muted-foreground mb-4">{error}</p>
   300→        <Button onClick={load} size="sm">
   301→          <RefreshCw className="h-3.5 w-3.5 mr-1.5" /> Retry
   302→        </Button>
   303→      </div>
   304→    )
   305→  }
   306→
   307→  if (!data) return null
   308→
   309→  const s = data.stats
   310→
   311→  return (
   312→    <div className="space-y-4">
   313→      {/* Top metric cards */}
   314→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   315→        <MetricCard
   316→          icon={<Users className="h-5 w-5" />}
   317→          label="Total Users"
   318→          value={s.users.total}
   319→          sub={`${s.users.active} active · ${s.users.banned} banned`}
   320→          color="text-blue-500"
   321→        />
   322→        <MetricCard
   323→          icon={<Activity className="h-5 w-5" />}
   324→          label="24h Volume"
   325→          value={formatUsd(s.trades.volume24hUsd)}
   326→          sub={`${s.trades.last24h} trades in 24h`}
   327→          color="text-green-500"
   328→        />
   329→        <MetricCard
   330→          icon={<Wallet className="h-5 w-5" />}
   331→          label="Total Balance"
   332→          value={formatUsd(s.wallets.totalUsdValue)}
   333→          sub={`${formatUsd(s.wallets.totalLockedValue)} locked in orders`}
   334→          color="text-yellow-500"
   335→        />
   336→        <MetricCard
   337→          icon={<Shield className="h-5 w-5" />}
   338→          label="KYC Pending"
   339→          value={s.users.kycPending}
   340→          sub={`${s.users.admins} managers`}
   341→          color="text-orange-500"
   342→        />
   343→      </div>
   344→
   345→      {/* Secondary stats */}
   346→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   347→        <SmallStat label="Active Pairs" value={`${s.pairs.active}/${s.pairs.total}`} />
   348→        <SmallStat label="Open Orders" value={s.orders.open} />
   349→        <SmallStat label="P2P Active" value={`${s.p2p.active}/${s.p2p.listings}`} />
   350→        <SmallStat label="Pending Tx" value={s.transactions.pending} />
   351→      </div>
   352→
   353→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   354→        {/* User growth chart */}
   355→        <div className="bg-card border border-border rounded-lg p-4">
   356→          <h3 className="font-semibold mb-3 flex items-center gap-2">
