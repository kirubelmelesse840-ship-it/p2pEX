   256→function AuthButtons() {

   264→  const [loading, setLoading] = useState(false)
   265→
   266→  const submit = async (mode: 'login' | 'signup') => {
   267→    if (mode === 'signup') {
   268→      if (password.length < 6) {
   269→        toast({ title: 'Password too short', description: 'At least 6 characters required', variant: 'destructive' })
   270→        return
   271→      }
   272→      if (password !== confirmPassword) {
   273→        toast({ title: 'Passwords do not match', variant: 'destructive' })
   274→        return
   275→      }
   276→    }

   278→    try {

   280→      const body = mode === 'login'

   284→        method: 'POST',
   285→        headers: { 'Content-Type': 'application/json' },
   286→        body: JSON.stringify(body),
   287→      })

   290→      setUser(data.user)

   292→        title: mode === 'login' ? 'Welcome back!' : 'Account created!',

   477→/**
   478→ * GoogleLoginDialog - Simulates the Google account chooser flow.
   479→ *
   480→ * 1. Shows Google accounts already signed into the device (from localStorage).
   481→ * 2. "Use another account" opens a signup form asking for email, password,
   482→ *    and confirm password — no email verification needed.
   483→ */
   484→function GoogleLoginDialog({ loading, onClose, onLogin }: {
   485→  loading: boolean
   486→  onClose: () => void
   487→  onLogin: (email: string, name?: string) => void
   488→}) {
   489→  const [step, setStep] = useState<'choose' | 'signup'>('choose')
   490→  const [email, setEmail] = useState('')
   491→  const [name, setName] = useState('')
   492→  const [password, setPassword] = useState('')
   493→  const [confirmPassword, setConfirmPassword] = useState('')
   494→  const [showPassword, setShowPassword] = useState(false)
   495→  const [submitting, setSubmitting] = useState(false)
   496→  const { toast } = useToast()
   497→
   498→  // Load detected Google accounts from localStorage
   499→  const [detectedAccounts, setDetectedAccounts] = useState<Array<{ email: string; name: string }>>([])
   500→
   501→  useEffect(() => {
   502→    try {
   503→      const stored = localStorage.getItem('google-signed-accounts')
   504→      if (stored) {
   505→        const accounts = JSON.parse(stored)
   506→        if (Array.isArray(accounts)) setDetectedAccounts(accounts)
   507→      }
   508→    } catch {}
   509→  }, [])
   510→
   511→  const pickAccount = (acc: { email: string; name: string }) => {
   512→    // Account already on device - log in directly
   513→    onLogin(acc.email, acc.name)
   514→  }
   515→
   516→  // Create a new account with email + password (no verification needed)
   517→  const createAccount = async () => {
   518→    if (!email.trim() || !password || !confirmPassword) {
   519→      toast({ title: 'All fields are required', variant: 'destructive' })
   520→      return
   521→    }
   522→    if (password.length < 6) {
   523→      toast({ title: 'Password too short', description: 'At least 6 characters required', variant: 'destructive' })
   524→      return
   525→    }
   526→    if (password !== confirmPassword) {
   527→      toast({ title: 'Passwords do not match', variant: 'destructive' })
   528→      return
   529→    }
   530→    setSubmitting(true)
   531→    try {
   532→      // Create the account via the signup API
   533→      const res = await fetch('/api/auth', {
   534→        method: 'POST',
   535→        headers: { 'Content-Type': 'application/json' },
   536→        body: JSON.stringify({
   537→          email: email.trim(),
   538→          password,
   539→          name: email.split('@')[0],
   540→        }),
   541→      })
   542→      const d = await res.json()
   543→      if (d.error) throw new Error(d.error)
   544→
   545→      // Save this account to detected accounts for next time
   546→      const newAcc = { email: email.trim(), name: d.user.name }
   547→      const updated = [...detectedAccounts.filter(a => a.email !== newAcc.email), newAcc]
   548→      setDetectedAccounts(updated)
   549→      try { localStorage.setItem('google-signed-accounts', JSON.stringify(updated)) } catch {}
   550→
   551→      toast({ title: 'Account created!', description: `Welcome to P2PEX, ${d.user.name}` })
   552→      // Call onLogin to set the user state (the signup already set the session cookie)
   553→      onLogin(email.trim(), d.user.name)
   554→    } catch (e: any) {
   555→      toast({ title: 'Sign up failed', description: e.message, variant: 'destructive' })
   556→    } finally {
   557→      setSubmitting(false)
   558→    }
   559→  }
   560→
   561→  return (
   562→    <Dialog open onOpenChange={onClose}>
   563→      <DialogContent className="max-w-md">
   564→        {/* Google-style header */}
   565→        <div className="flex flex-col items-center text-center mb-5">
   566→          <div className="mb-3">
   567→            <svg className="h-10 w-10" viewBox="0 0 24 24" xmlns="http://www.w3.org/2000/svg">
   568→              <path fill="#4285F4" d="M22.56 12.25c0-.78-.07-1.53-.2-2.25H12v4.26h5.92c-.26 1.37-1.04 2.53-2.21 3.31v2.77h3.57c2.08-1.92 3.28-4.74 3.28-8.09z"/>
   569→              <path fill="#34A853" d="M12 23c2.97 0 5.46-.98 7.28-2.66l-3.57-2.77c-.98.66-2.23 1.06-3.71 1.06-2.86 0-5.29-1.93-6.16-4.53H2.18v2.84C3.99 20.53 7.7 23 12 23z"/>
   570→              <path fill="#FBBC05" d="M5.84 14.09c-.22-.66-.35-1.36-.35-2.09s.13-1.43.35-2.09V7.07H2.18C1.43 8.55 1 10.22 1 12s.43 3.45 1.18 4.93l2.85-2.22.81-.62z"/>
   571→              <path fill="#EA4335" d="M12 5.38c1.62 0 3.06.56 4.21 1.64l3.15-3.15C17.45 2.09 14.97 1 12 1 7.7 1 3.99 3.47 2.18 7.07l3.66 2.84c.87-2.6 3.3-4.53 6.16-4.53z"/>
   572→            </svg>
   573→          </div>
   574→          <h2 className="text-lg font-bold">
   575→            {step === 'signup' ? 'Create your account' : 'Sign in with Google'}
   576→          </h2>
   577→          <p className="text-xs text-muted-foreground mt-1">
   578→            {step === 'choose' && 'Choose an account to continue to P2PEX'}
   579→            {step === 'signup' && 'Enter your details to create a new account'}
   580→          </p>
   581→        </div>
   582→
   583→        {/* STEP 1: Choose account (detected accounts on device) */}
   584→        {step === 'choose' && (
   585→          <div className="space-y-2">
   586→            {detectedAccounts.length > 0 && (
   587→              <>
   588→                <div className="text-xs text-muted-foreground px-1 mb-1">
   589→                  Accounts on this device:
   590→                </div>
   591→                {detectedAccounts.map(acc => (
   592→                  <button
   593→                    key={acc.email}
   594→                    onClick={() => pickAccount(acc)}
   595→                    disabled={loading}
   596→                    className="w-full flex items-center gap-3 p-3 rounded-lg border border-border hover:border-primary/50 hover:bg-muted/30 transition text-left disabled:opacity-50"
   597→                  >
   598→                    <div className="flex h-9 w-9 items-center justify-center rounded-full bg-gradient-to-br from-blue-500 to-purple-500 text-white text-sm font-bold flex-shrink-0">
   599→                      {acc.name.slice(0, 1).toUpperCase() || acc.email.slice(0, 1).toUpperCase()}
   600→                    </div>
   601→                    <div className="flex-1 min-w-0">
   602→                      <div className="text-sm font-medium truncate">{acc.name}</div>
   603→                      <div className="text-xs text-muted-foreground truncate">{acc.email}</div>
   604→                    </div>
   605→                  </button>
   606→                ))}
   607→                <div className="border-t border-border my-2" />
   608→              </>
   609→            )}
   610→
   611→            <button
   612→              onClick={() => setStep('signup')}
   613→              disabled={loading}
   614→              className="w-full flex items-center gap-3 p-3 rounded-lg border border-border hover:border-primary/50 hover:bg-muted/30 transition text-left disabled:opacity-50"
   615→            >
   616→              <div className="flex h-9 w-9 items-center justify-center rounded-full border border-dashed border-muted-foreground text-muted-foreground flex-shrink-0">
   617→                <Plus className="h-4 w-4" />
   618→              </div>
   619→              <div className="text-sm font-medium">Use another account</div>
   620→            </button>
   621→
   622→            <div className="text-xs text-muted-foreground bg-muted/30 rounded p-2 flex items-start gap-2 mt-3">
   623→              <Shield className="h-3 w-3 mt-0.5 flex-shrink-0" />
   624→              <span>
   625→                P2PEX will use your email to create or access your account.
   626→                We don't access your Google data or store your Google password.
   627→              </span>
   628→            </div>
   629→          </div>
   630→        )}
   631→
   632→        {/* STEP 2: Sign up with email + password */}
   633→        {step === 'signup' && (
   634→          <div className="space-y-3">
   635→            <div>
   636→              <label className="text-xs font-medium text-muted-foreground flex items-center gap-1">
   637→                <Mail className="h-3 w-3" /> Email
   638→              </label>
   639→              <Input
   640→                type="email"
   641→                value={email}
   642→                onChange={e => setEmail(e.target.value)}
   643→                placeholder="your.email@gmail.com"
   644→                className="mt-1"
   645→                autoFocus
   646→              />
   647→            </div>
   648→
   649→            <div>
   650→              <label className="text-xs font-medium text-muted-foreground">Password</label>
   651→              <div className="relative">
   652→                <Input
   653→                  type={showPassword ? 'text' : 'password'}
   654→                  value={password}
   655→                  onChange={e => setPassword(e.target.value)}
   656→                  placeholder="••••••••"
   657→                  className="mt-1 pr-10"
   658→                  onKeyDown={e => { if (e.key === 'Enter') createAccount() }}
   659→                />
   660→                <button
   661→                  onClick={() => setShowPassword(!showPassword)}
   662→                  className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   663→                >
   664→                  {showPassword ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   665→                </button>
   666→              </div>
   667→              <p className="text-xs text-muted-foreground mt-1">Minimum 6 characters</p>
   668→            </div>
   669→
   670→            <div>
   671→              <label className="text-xs font-medium text-muted-foreground">Confirm Password</label>
   672→              <Input
   673→                type={showPassword ? 'text' : 'password'}
   674→                value={confirmPassword}
   675→                onChange={e => setConfirmPassword(e.target.value)}
   676→                placeholder="••••••••"
   677→                className={`mt-1 ${confirmPassword && confirmPassword !== password ? 'border-red-500' : confirmPassword && confirmPassword === password ? 'border-green-500' : ''}`}
   678→                onKeyDown={e => { if (e.key === 'Enter') createAccount() }}
   679→              />
   680→              {confirmPassword && confirmPassword !== password && (
   681→                <p className="text-xs text-red-500 mt-1">Passwords do not match</p>
   682→              )}
   683→              {confirmPassword && confirmPassword === password && (
   684→                <p className="text-xs text-green-500 mt-1">✓ Passwords match</p>
   685→              )}
   686→            </div>
   687→
   688→            <Button
   689→              className="w-full bg-[#4285F4] hover:bg-[#3367D6] text-white"
   690→              onClick={createAccount}
   691→              disabled={submitting || !email.trim() || !password || !confirmPassword || password !== confirmPassword || password.length < 6}
   692→            >
   693→              {submitting ? 'Creating account...' : 'Create Account'}
   694→            </Button>
   695→
   696→            <button
   697→              className="w-full text-center text-xs text-muted-foreground hover:text-foreground"
   698→              onClick={() => setStep('choose')}
   699→            >
   700→              ← Back to account list
   701→            </button>
   702→          </div>
   703→        )}
   704→      </DialogContent>
   705→    </Dialog>
   706→  )

   709→// =================== USER NOTIFICATION BELL ===================
   710→function UserNotificationBell() {
   711→  const [notifications, setNotifications] = useState<any[]>([])
   712→  const [unreadCount, setUnreadCount] = useState(0)
   713→  const [open, setOpen] = useState(false)
   714→
   715→  const load = useCallback(async () => {
   716→    try {
   717→      const res = await fetch('/api/notifications')
   718→      const d = await res.json()
   719→      if (d.error) return
   720→      setNotifications(d.notifications || [])
   721→      setUnreadCount(d.unreadCount || 0)
   722→    } catch {}
   723→  }, [])
   724→
   725→  useEffect(() => {
   726→    load()
   727→    const t = setInterval(load, 15000)
   728→    return () => clearInterval(t)
   729→  }, [load])
   730→
   731→  const markRead = async (id: string) => {
   732→    try {
   733→      await fetch('/api/notifications', {
   734→        method: 'POST',
   735→        headers: { 'Content-Type': 'application/json' },
   736→        body: JSON.stringify({ notificationId: id }),
   737→      })
   738→      load()
   739→    } catch {}
   740→  }
   741→
   742→  const markAllRead = async () => {
   743→    for (const n of notifications.filter(n => !n.isRead)) {
   744→      await markRead(n.id)
   745→    }
   746→  }
   747→
   748→  const formatTimeAgo = (time: string) => {
   749→    const diff = Date.now() - new Date(time).getTime()
   750→    if (diff < 60000) return 'just now'
   751→    if (diff < 3600000) return Math.floor(diff / 60000) + 'm ago'
   752→    if (diff < 86400000) return Math.floor(diff / 3600000) + 'h ago'
   753→    return Math.floor(diff / 86400000) + 'd ago'
   754→  }
   755→
   756→  const colorForType = (type: string) => {
   757→    switch (type) {
   758→      case 'success': return 'text-green-500'
   759→      case 'warning': return 'text-yellow-500'
   760→      case 'announcement': return 'text-purple-500'
   761→      default: return 'text-blue-500'
   762→    }
   763→  }
   764→
   765→  return (
   766→    <div className="relative">
   767→      <Button
   768→        variant="ghost"
   769→        size="icon"
   770→        className="relative"
   771→        onClick={() => { setOpen(!open); if (!open) load() }}
   772→        title="Notifications"
   773→      >
   774→        <Bell className="h-4 w-4" />
   775→        {unreadCount > 0 && (
   776→          <span className="absolute -top-0.5 -right-0.5 flex h-3.5 min-w-3.5 items-center justify-center rounded-full bg-red-500 text-white text-[9px] font-bold px-1">
   777→            {unreadCount > 9 ? '9+' : unreadCount}
   778→          </span>
   779→        )}
   780→      </Button>
   781→
   782→      {open && (
   783→        <>
   784→          <div className="fixed inset-0 z-40" onClick={() => setOpen(false)} />
   785→          <div className="absolute right-0 top-full mt-1 w-80 max-h-96 overflow-y-auto bg-card border border-border rounded-lg shadow-xl z-50">
   786→            <div className="sticky top-0 bg-card border-b border-border px-3 py-2 flex items-center justify-between">
   787→              <span className="text-sm font-semibold">Notifications</span>
   788→              {unreadCount > 0 && (
   789→                <button className="text-xs text-primary hover:underline" onClick={markAllRead}>
   790→                  Mark all read
   791→                </button>
   792→              )}
   793→            </div>
   794→
   795→            {notifications.length === 0 ? (
   796→              <div className="p-6 text-center text-xs text-muted-foreground">
   797→                <Bell className="h-8 w-8 mx-auto mb-2 opacity-30" />
   798→                No notifications
   799→              </div>
   800→            ) : (
   801→              <div className="divide-y divide-border/40">
   802→                {notifications.map(n => (
   803→                  <div
   804→                    key={n.id}
   805→                    className={'p-3 hover:bg-muted/30 transition ' + (!n.isRead ? 'bg-primary/5' : '')}
   806→                    onClick={() => { if (!n.isRead) markRead(n.id) }}
   807→                  >
   808→                    <div className="flex items-start gap-2">
   809→                      <div className={'flex-shrink-0 mt-0.5 ' + colorForType(n.type)}>
   810→                        <Bell className="h-4 w-4" />
   811→                      </div>
   812→                      <div className="flex-1 min-w-0">
   813→                        <div className="flex items-center gap-1">
   814→                          <span className="text-xs font-medium">{n.title}</span>
   815→                          {!n.isRead && <span className="w-1.5 h-1.5 rounded-full bg-blue-500 flex-shrink-0" />}
   816→                        </div>
   817→                        <p className="text-xs text-muted-foreground mt-0.5">{n.message}</p>
   818→                        <p className="text-[10px] text-muted-foreground mt-1">{formatTimeAgo(n.createdAt)}</p>
   819→                      </div>
   820→                    </div>
   821→                  </div>
   822→                ))}
   823→              </div>
   824→            )}
   825→          </div>
   826→        </>
   827→      )}
   828→    </div>
   829→  )
   830→}
