    89→    89→export function P2PView() {
    90→    90→  const { user, setView } = useAppStore()
    91→    91→  const { toast } = useToast()
    92→    92→  const [listings, setListings] = useState<Listing[]>([])
    93→    93→  const [orders, setOrders] = useState<P2POrder[]>([])
    94→    94→  const [loading, setLoading] = useState(true)
    95→    95→  const [tab, setTab] = useState<'buy' | 'sell' | 'orders' | 'create'>('buy')
    96→    96→  const [filters, setFilters] = useState({ asset: 'USDT', fiat: 'ETB', paymentMethod: 'ALL' })
    97→    97→  const [tradeDialog, setTradeDialog] = useState<Listing | null>(null)
    98→    98→  const [orderDialog, setOrderDialog] = useState<P2POrder | null>(null)
    99→    99→  const [createOpen, setCreateOpen] = useState(false)
   100→   100→
   101→   101→  const [prevOrderStatuses, setPrevOrderStatuses] = useState<Record<string, string>>({})
   102→   102→
   103→   103→  const load = useCallback(async () => {
   104→   104→    try {
   105→   105→      const [listRes, ordersRes] = await Promise.all([
   106→   106→        fetch(`/api/p2p/listings?asset=${filters.asset}&fiat=${filters.fiat}`),
   107→   107→        user ? fetch('/api/p2p/orders?role=all') : Promise.resolve(null),
   108→   108→      ])
   109→   109→      const listData = await listRes.json()
   110→   110→      setListings(listData.listings || [])
   111→   111→      if (ordersRes) {
   112→   112→        const ordersData = await ordersRes.json()
   113→   113→        const newOrders = ordersData.orders || []
   114→   114→        // Detect status changes and notify the user
   115→   115→        for (const o of newOrders) {
   116→   116→          const prevStatus = prevOrderStatuses[o.id]
   117→   117→          if (prevStatus && prevStatus !== o.status) {
   118→   118→            // Status changed — show notification
   119→   119→            if (o.status === 'PAID' && o.myRole === 'BUYER') {
   120→   120→              toast({ title: '✅ Payment Approved!', description: `Your payment for ${o.amount} ${o.asset} has been verified. The seller will release your crypto shortly.` })
   121→   121→            } else if (o.status === 'PAID' && o.myRole === 'SELLER') {
   122→   122→              toast({ title: '💰 Payment Verified', description: `The buyer's payment for ${o.amount} ${o.asset} has been verified. You can now release the crypto.` })
   123→   123→            } else if (o.status === 'CANCELED' && o.myRole === 'BUYER') {
   124→   124→              toast({ title: '❌ Payment Rejected', description: `Your payment for ${o.amount} ${o.asset} was rejected. The order has been canceled. You can place a new order.`, variant: 'destructive' })
   125→   125→            } else if (o.status === 'CANCELED' && o.myRole === 'SELLER') {
   126→   126→              toast({ title: '❌ Order Canceled', description: `The order for ${o.amount} ${o.asset} has been canceled.`, variant: 'destructive' })
   127→   127→            } else if (o.status === 'COMPLETED') {
   128→   128→              toast({ title: '🎉 Trade Completed!', description: `Your trade of ${o.amount} ${o.asset} has been completed successfully.` })
   129→   129→            }
   130→   130→          }
   131→   131→        }
   132→   132→        // Update previous statuses
   133→   133→        const newStatuses: Record<string, string> = {}
   134→   134→        for (const o of newOrders) newStatuses[o.id] = o.status
   135→   135→        setPrevOrderStatuses(newStatuses)
   136→   136→        setOrders(newOrders)
   137→   137→      }
   138→   138→    } catch (e) {
   139→   139→      console.error(e)
   140→   140→    } finally {
   141→   141→      setLoading(false)
   142→   142→    }
   143→   143→  }, [filters.asset, filters.fiat, user, prevOrderStatuses, toast])
   144→   144→
   145→   145→  useEffect(() => { load() }, [load])
   146→   146→  useEffect(() => {
   147→   147→    if (!user) return
   148→   148→    // Poll every 5 seconds for faster status updates
   149→   149→    const t = setInterval(load, 5000)
   150→   150→    return () => clearInterval(t)
   151→   151→  }, [user, load])
   152→   152→
   153→   153→  // Reset payment method filter when fiat changes
   154→   154→  useEffect(() => {
   155→   155→    setFilters(f => ({ ...f, paymentMethod: 'ALL' }))
   156→   156→  }, [filters.fiat])
   157→   157→
   158→   158→  const buyListings = listings.filter(l => l.side === 'SELL') // user wants to BUY, so look for SELL listings
   159→   159→  const sellListings = listings.filter(l => l.side === 'BUY')
   160→   160→
   161→   161→  // Apply payment method filter (matches listings that include the selected method)
   162→   162→  const filterByPayment = (list: Listing[]) => {
   163→   163→    if (filters.paymentMethod === 'ALL') return list
   164→   164→    return list.filter(l => l.paymentMethods.includes(filters.paymentMethod))
   165→   165→  }
   166→   166→
   167→   167→  const displayed = tab === 'buy' ? filterByPayment(buyListings) : tab === 'sell' ? filterByPayment(sellListings) : []
   168→   168→
   169→   169→  // Available payment methods for the selected fiat currency
   170→   170→  const availablePaymentMethods = FIAT_PAYMENT_METHODS[filters.fiat] || ['Bank Transfer']
   171→   171→
   172→   172→  return (
   173→   173→    <div className="container mx-auto px-3 sm:px-4 py-4 max-w-6xl">
   174→   174→      <BackButton to="home" />
   175→   175→
   176→   176→      <div className="flex items-center justify-between mb-4 mt-1">
   177→   177→        <div>
   178→   178→          <h1 className="text-2xl font-bold flex items-center gap-2">
   179→   179→            <Users className="h-6 w-6" />
   180→   180→            P2P Marketplace
   181→   181→          </h1>
   182→   182→          <p className="text-sm text-muted-foreground">Trade crypto directly with other users in your local currency</p>
   183→   183→        </div>
   184→   184→        {user && (
   185→   185→          <Button onClick={() => setCreateOpen(true)} className="hidden sm:flex">
   186→   186→            <Plus className="h-4 w-4 mr-1.5" /> Post Ad
   187→   187→          </Button>
   188→   188→        )}
   189→   189→      </div>
   190→   190→
   191→   191→      {/* Filter bar */}
   192→   192→      <div className="flex flex-wrap items-center gap-2 mb-4 p-3 bg-card rounded-lg border border-border">
   193→   193→        <div className="flex items-center gap-1">
   194→   194→          <Button
   195→   195→            variant={tab === 'buy' ? 'default' : 'outline'}
   196→   196→            size="sm"
   197→   197→            className={tab === 'buy' ? 'bg-green-500 hover:bg-green-600 text-white' : ''}
   198→   198→            onClick={() => setTab('buy')}
   199→   199→          >
   200→   200→            Buy
   201→   201→          </Button>
   202→   202→          <Button
   203→   203→            variant={tab === 'sell' ? 'default' : 'outline'}
   204→   204→            size="sm"
   205→   205→            className={tab === 'sell' ? 'bg-red-500 hover:bg-red-600 text-white' : ''}
   206→   206→            onClick={() => setTab('sell')}
   207→   207→          >
   208→   208→            Sell
   209→   209→          </Button>
   210→   210→          {user && (
   211→   211→            <Button
   212→   212→              variant={tab === 'orders' ? 'secondary' : 'ghost'}
   213→   213→              size="sm"
   214→   214→              onClick={() => setTab('orders')}
   215→   215→            >
   216→   216→              My Orders ({orders.filter(o => o.status !== 'COMPLETED' && o.status !== 'CANCELED').length})
   217→   217→            </Button>
   218→   218→          )}
   219→   219→        </div>
   220→   220→
   221→   221→        <div className="ml-auto flex flex-wrap items-center gap-2">
   222→   222→          <Select value={filters.asset} onValueChange={(v) => setFilters(f => ({ ...f, asset: v }))}>
   223→   223→            <SelectTrigger className="w-24 h-8"><SelectValue /></SelectTrigger>
   224→   224→            <SelectContent>
   225→   225→              {CRYPTO_ASSETS.map(a => <SelectItem key={a} value={a}>{a}</SelectItem>)}
   226→   226→            </SelectContent>
   227→   227→          </Select>
   228→   228→          <Select value={filters.fiat} onValueChange={(v) => setFilters(f => ({ ...f, fiat: v }))}>
   229→   229→            <SelectTrigger className="w-24 h-8"><SelectValue /></SelectTrigger>
   230→   230→            <SelectContent>
   231→   231→              {FIAT_CURRENCIES.map(f => <SelectItem key={f} value={f}>{f}</SelectItem>)}
   232→   232→            </SelectContent>
   233→   233→          </Select>
   234→   234→          <Select value={filters.paymentMethod} onValueChange={(v) => setFilters(f => ({ ...f, paymentMethod: v }))}>
   235→   235→            <SelectTrigger className="w-32 h-8"><SelectValue placeholder="Payment" /></SelectTrigger>
   236→   236→            <SelectContent>
   237→   237→              <SelectItem value="ALL">All Methods</SelectItem>
   238→   238→              {availablePaymentMethods.map(m => <SelectItem key={m} value={m}>{m}</SelectItem>)}
   239→   239→            </SelectContent>
   240→   240→          </Select>
   241→   241→        </div>
   242→   242→      </div>
   243→   243→
   244→   244→      {/* Listings grid */}
   245→   245→      {tab === 'buy' || tab === 'sell' ? (
   246→   246→        loading ? (
   247→   247→          <div className="text-center py-12 text-muted-foreground">Loading listings...</div>
   248→   248→        ) : displayed.length === 0 ? (
   249→   249→          <div className="text-center py-12">
   250→   250→            <Users className="h-12 w-12 mx-auto text-muted-foreground mb-3" />
   251→   251→            <p className="text-muted-foreground">No {tab === 'buy' ? 'sellers' : 'buyers'} available for {filters.asset}/{filters.fiat}</p>
   252→   252→            {user && (
   253→   253→              <Button className="mt-4" onClick={() => setCreateOpen(true)}>
   254→   254→                <Plus className="h-4 w-4 mr-1.5" /> Post the first ad
   255→   255→              </Button>
   256→   256→            )}
   257→   257→          </div>
   258→   258→        ) : (
   259→   259→          <div className="grid gap-2">
   260→   260→            {displayed.map(l => (
   261→   261→              <ListingCard key={l.id} listing={l} onTrade={() => setTradeDialog(l)} />
   262→   262→            ))}
   263→   263→          </div>
   264→   264→        )
   265→   265→      ) : null}
   266→   266→
   267→   267→      {/* My Orders */}
   268→   268→      {tab === 'orders' && (
   269→   269→        <div className="space-y-2">
   270→   270→          {!user ? (
   271→   271→            <div className="text-center py-12">
   272→   272→              <p className="text-muted-foreground mb-3">Log in to view your orders</p>
   273→   273→              <Button onClick={() => setView('home')}>Back to Home</Button>
   274→   274→            </div>
   275→   275→          ) : orders.length === 0 ? (
   276→   276→            <div className="text-center py-12 text-muted-foreground">
   277→   277→              <Clock className="h-12 w-12 mx-auto mb-3" />
   278→   278→              <p>No P2P orders yet</p>
   279→   279→            </div>
   280→   280→          ) : (
   281→   281→            orders.map(o => <OrderCard key={o.id} order={o} onClick={() => setOrderDialog(o)} />)
   282→   282→          )}
   283→   283→        </div>
   284→   284→      )}
   285→   285→
   286→   286→      {/* Trade dialog */}
   287→   287→      {tradeDialog && (
   288→   288→        <TradeDialog
   289→   289→          listing={tradeDialog}
   290→   290→          onClose={() => setTradeDialog(null)}
   291→   291→          onSuccess={() => { load(); setTab('orders') }}
   292→   292→        />
   293→   293→      )}
   294→   294→
   295→   295→      {/* Order detail dialog */}
   296→   296→      {orderDialog && (
   297→   297→        <OrderDialog
   298→   298→          order={orderDialog}
   299→   299→          onClose={() => setOrderDialog(null)}
   300→   300→          onSuccess={() => { load(); setOrderDialog(null) }}
   301→   301→        />
   302→   302→      )}
   303→   303→
   304→   304→      {/* Create listing dialog */}
   305→   305→      {createOpen && user && (
   306→   306→        <CreateListingDialog
   307→   307→          onClose={() => setCreateOpen(false)}
   308→   308→          onSuccess={() => { load(); setCreateOpen(false) }}
   309→   309→        />
   310→   310→      )}
   311→   311→    </div>
   312→   312→  )

   315→   315→function ListingCard({ listing, onTrade }: { listing: Listing; onTrade: () => void }) {
   316→   316→  const { user } = useAppStore()
   317→   317→  const isMine = user && listing.user.name === user.name
   318→   318→  const hasVerification = listing.user.kycVerified
   319→   319→  const kycLevel = listing.user.kycLevel || 0
   320→   320→  // Use real trades count and rating from the listing (set by admin)
   321→   321→  const tradesCount = listing.tradesCount || 0
   322→   322→  const rating = listing.rating || 4.9
   323→   323→
   324→   324→  return (
   325→   325→    <div className="bg-card border border-border rounded-lg p-4 hover:border-primary/50 transition">
   326→   326→      <div className="grid grid-cols-2 md:grid-cols-5 gap-3 items-center">
   327→   327→        {/* Advertiser with verification status */}
   328→   328→        <div className="flex items-center gap-2">
   329→   329→          <div className="flex h-9 w-9 items-center justify-center rounded-full bg-primary/10 text-primary font-medium text-sm">
   330→   330→            {listing.user.name.slice(0, 2).toUpperCase()}
   331→   331→          </div>
   332→   332→          <div>
   333→   333→            <div className="font-medium text-sm flex items-center gap-1">
   334→   334→              {listing.user.name}
   335→   335→              {hasVerification && (
   336→   336→                <span
   337→   337→                  className={`inline-flex items-center gap-0.5 text-[10px] px-1 py-0.5 rounded ${
   338→   338→                    kycLevel >= 2
   339→   339→                      ? 'bg-green-500/15 text-green-600 dark:text-green-400'
   340→   340→                      : 'bg-yellow-500/15 text-yellow-600 dark:text-yellow-400'
   341→   341→                  }`}
   342→   342→                  title={`KYC Level ${kycLevel} verified`}
   343→   343→                >
   344→   344→                  <BadgeCheck className="h-3 w-3" />
   345→   345→                  L{kycLevel}
   346→   346→                </span>
   347→   347→              )}
   348→   348→            </div>
   349→   349→            <div className="text-xs text-muted-foreground flex items-center gap-1">
   350→   350→              {hasVerification ? (
   351→   351→                <span className="flex items-center gap-0.5 text-green-600 dark:text-green-400">
   352→   352→                  <Shield className="h-3 w-3" /> Verified
   353→   353→                </span>
   354→   354→              ) : (
   355→   355→                <span className="text-muted-foreground">Unverified</span>
   356→   356→              )}
   357→   357→              <span className="mx-1">·</span>
   358→   358→              <Star className="h-3 w-3 fill-yellow-500 text-yellow-500" />
   359→   359→              <span>{rating.toFixed(1)}</span>
   360→   360→              <span className="mx-1">·</span>
   361→   361→              <span>{tradesCount} trades</span>
   362→   362→            </div>
   363→   363→          </div>
   364→   364→        </div>
   365→   365→
   366→   366→        {/* Price */}
   367→   367→        <div>
   368→   368→          <div className="text-xs text-muted-foreground">Price</div>
   369→   369→          <div className="font-bold tabular-nums">
   370→   370→            {formatPrice(listing.price)} {listing.fiatCurrency}
   371→   371→          </div>
   372→   372→        </div>
   373→   373→
   374→   374→        {/* Limit */}
   375→   375→        <div className="hidden md:block">
   376→   376→          <div className="text-xs text-muted-foreground">Limit</div>
   377→   377→          <div className="text-sm tabular-nums">
   378→   378→            {formatPrice(listing.minOrder)} - {formatPrice(listing.maxOrder)}
   379→   379→          </div>
   380→   380→        </div>
   381→   381→
   382→   382→        {/* Payment - show all methods with matched highlight */}
   383→   383→        <div className="hidden md:block">
   384→   384→          <div className="text-xs text-muted-foreground">Payment</div>
   385→   385→          <div className="flex flex-wrap gap-1 mt-0.5">
   386→   386→            {listing.paymentMethods.slice(0, 3).map(m => (
   387→   387→              <Badge key={m} variant="secondary" className="text-[10px] py-0 px-1.5">{m}</Badge>
   388→   388→            ))}
   389→   389→            {listing.paymentMethods.length > 3 && (
   390→   390→              <span className="text-[10px] text-muted-foreground">+{listing.paymentMethods.length - 3}</span>
   391→   391→            )}
   392→   392→          </div>
   393→   393→        </div>
   394→   394→
   395→   395→        {/* Action */}
   396→   396→        <div className="flex justify-end">
   397→   397→          <Button
   398→   398→            variant={listing.side === 'SELL' ? 'default' : 'outline'}
   399→   399→            className={listing.side === 'SELL' ? 'bg-green-500 hover:bg-green-600 text-white' : 'border-red-500 text-red-500 hover:bg-red-500/10'}
   400→   400→            onClick={onTrade}
   401→   401→            disabled={!!isMine}
   402→   402→          >
   403→   403→            {listing.side === 'SELL' ? 'Buy' : 'Sell'} {listing.asset}

   405→   405→        </div>
   406→   406→      </div>
   407→   407→
   408→   408→      {/* Available amount */}

   410→   410→        <span>Available: {formatQty(listing.available)} {listing.asset}</span>
   411→   411→        <span>Trade limit: {formatPrice(listing.minOrder)} - {formatPrice(listing.maxOrder)} {listing.fiatCurrency}</span>
   412→   412→      </div>
   413→   413→    </div>
   414→   414→  )
